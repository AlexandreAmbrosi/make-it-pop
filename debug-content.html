<script lang="ts">
  import { ArrowUpRight } from 'lucide-svelte'
  import { page } from '$app/stores'
  import { onMount } from 'svelte'
  import gsap from 'gsap'
  import { ScrollTrigger } from 'gsap/dist/ScrollTrigger'

  onMount(() => {
    gsap.registerPlugin(ScrollTrigger)

    const showAnim = gsap
      .from('#homepage-navbar', {
        yPercent: -130, // Just enough to hide
        paused: true,
        duration: 0.3, // Standard speed
        ease: 'power2.out', // Smooth standard easing
      })
      .progress(1)

    // Show/Hide on Scroll Direction (Delayed until after Hero)
    ScrollTrigger.create({
      start: '500 top', // Trigger after 500px scroll
      end: 'max',
      onUpdate: (self) => {
        self.direction === -1 ? showAnim.play() : showAnim.reverse()
      },
    })
